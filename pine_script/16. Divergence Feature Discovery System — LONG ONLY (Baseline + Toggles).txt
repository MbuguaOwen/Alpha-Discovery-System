f_inRangeHM(hm_val, startHM, endHM) =>
    // Supports wrap-around (e.g. crossing midnight)
    startHM <= endHM ? (hm_val >= startHM and hm_val <= endHM) : (hm_val >= startHM or hm_val <= endHM)

f_inSessionAt(t) =>
    // Bypass when gate is OFF, or when gate is ON but user forgot to tick any session.
    if (not useSessionGate) or (useSessionGate and not anySessEnabled)
        true
    else
        int h  = hour(t, tzSess)
        int m  = minute(t, tzSess)
        int hm = h * 100 + m
        bool nyOk     = useNY     and f_inRangeHM(hm, 1630, 2300)
        bool londonOk = useLondon and f_inRangeHM(hm, 1100, 1930)
        bool tokyoOk  = useTokyo  and f_inRangeHM(hm,  300, 1200)
        bool sydneyOk = useSydney and f_inRangeHM(hm,  100, 1000)
        nyOk or londonOk or tokyoOk or sydneyOk

// Rolling sum using SMA * len (portable)
f_rsum(x, len) =>
    ta.sma(x, len) * len

// True Range (portable)
f_tr() =>
    float a = high - low
    float b = math.abs(high - close[1])
    float c = math.abs(low  - close[1])
    math.max(a, math.max(b, c))

// Wilder DMI/ADX (portable)
f_dmi_adx(_len) =>
    upMove   = high - high[1]
    downMove = low[1] - low

    plusDM  = (upMove > downMove and upMove > 0) ? upMove : 0.0
    minusDM = (downMove > upMove and downMove > 0) ? downMove : 0.0

    tr  = f_tr()
    atr = ta.rma(tr, _len)

    pDM = ta.rma(plusDM, _len)
    mDM = ta.rma(minusDM, _len)

    pDI = atr > 0 ? 100.0 * (pDM / atr) : 0.0
    mDI = atr > 0 ? 100.0 * (mDM / atr) : 0.0

    sumDI = pDI + mDI
    dx = sumDI > 0 ? 100.0 * math.abs(pDI - mDI) / sumDI : 0.0
    adx = ta.rma(dx, _len)

    [pDI, mDI, adx]

w_weighted_vol(i, baseVol) =>

w_inRow(rowTop, rowBot, hi, lo) =>
    not (lo > rowTop or hi < rowBot)

w_resetArrays() =>
    array.clear(w_bullVol)
    array.clear(w_totVol)
    for r0 = 0 to w_rowCount - 1
        array.push(w_bullVol, 0.0)
        array.push(w_totVol,  0.0)

    w_vpBot := w_vpl
    w_step  := (w_vpTop - w_vpBot) / w_rowCount
    w_resetArrays()

    for i = 0 to w_analyzeBars
        hi = high[i]
        lo = low[i]
        vv = w_weighted_vol(i, volume[i])
        for r = 0 to w_rowCount - 1
            rowTop = w_vpTop - w_step * r
            rowBot = rowTop - w_step
            if w_inRow(rowTop, rowBot, hi, lo)
                if close[i] > open[i]
                    array.set(w_bullVol, r, array.get(w_bullVol, r) + vv)
                array.set(w_totVol,  r, array.get(w_totVol,  r) + vv)

    bullPctAtPrice := na
    if not na(w_step) and w_step != 0
        for r = 0 to w_rowCount - 1
            rowTop = w_vpTop - w_step * r
            rowBot = rowTop - w_step
            if close <= rowTop and close >= rowBot
                tv = array.get(w_totVol, r)
                bv = array.get(w_bullVol, r)

        g_pivotGap    = (not useMinPivotGap) or (barsFromLastPivot >= minPivotGapBars)
        g_classic     = (not classicOnly) or (divType == "Classic")

        g_volPivot    = (not useVolPivotGate) or (volRatioAtPivot >= minVolRatioPivot)
        g_volEntry    = (not useVolEntryGate) or (volAtEntry >= minVolRatioEntry)

        g_atrRatio    = (not useAtrRatioGate) or (atrAtEntry <= atrRatioMax)

        g_regime      = dailyEmaOk and dailySlopeOk and dailyAdxOk and dailyDiOk
        g_session     = (not useSessionGate) or sessionOkAtPivot
        g_flow        = cvdOk and wv70Ok and volBehaviorPass and volSpikeOk

        pivotPass = g_oscStrength and g_pivotGap and g_classic and g_volPivot and g_volEntry and g_atrRatio and g_regime and g_session and g_flow

        tradable = (tradeMode == "BASELINE_ALL") ? true : pivotPass

        if mode == "TRADE" and tradable and cooldownOk and (not oneTradeAtATime or strategy.position_size == 0)
            if entryMode == "Raw"
                longSignal := true
            else
                longSetup  := true
                setupPass  := tradable
                setupPivot := pl_price
                setupBar   := bar_index
                longTrig   := bosHighest
                totalSetups += 1

                if showZones
                    if not na(zoneBox)
                        box.delete(zoneBox)
                    zH = atr * 1.0
                    zoneBox := box.new(
                        left=bar_index,
                        top=setupPivot + zH,
                        right=bar_index + maxWaitBars,
                        bottom=setupPivot - zH,
                        bgcolor=color.new(color.teal, 88),
                        border_color=color.new(color.teal, 50)
                    )

    // update pivot memory always
    lastPL_price := pl_price
    lastPL_osc   := pl_osc
    lastPL_bar   := pl_bar

// Confirm trigger
if mode == "TRADE" and entryMode == "Confirm(BOS)" and longSetup
    if (bar_index - setupBar) > maxWaitBars
        longSetup := false
        setupPass := false
        if not na(zoneBox)
            box.delete(zoneBox)
            zoneBox := na
    else
        buf    = atr * bosAtrBuffer
        trigOk = useBOSConfirm ? (close > (longTrig + buf)) : (close > open)

        triggerFlowOk    = cvdOk and wv70Ok and volBehaviorPass and volSpikeOk
        triggerSessionOk = (not useSessionGate) or sessionOk

        finalFlowOk    = (tradeMode == "BASELINE_ALL") ? true : triggerFlowOk
        finalSessionOk = (tradeMode == "BASELINE_ALL") ? true : triggerSessionOk

        if okBar and warmed and cooldownOk and setupPass and trigOk and finalFlowOk and finalSessionOk and (not oneTradeAtATime or strategy.position_size == 0)
            longSignal := true
            longSetup := false
            setupPass := false
            if not na(zoneBox)
                box.delete(zoneBox)
                zoneBox := na

//═════════════════════════════════════════════════════════════
// EXECUTION (LONG ONLY) + standardized exits
//═════════════════════════════════════════════════════════════
if longSignal
    strategy.entry("LONG", strategy.long)

entered = (strategy.position_size > 0) and (strategy.position_size[1] == 0)
if entered
    lastEntryBar := bar_index
    totalEntries += 1

var float atrEntry = na
if entered
    atrEntry := atr
if strategy.position_size == 0
    atrEntry := na

if strategy.position_size > 0 and not na(atrEntry)
    ep = strategy.position_avg_price
    sl = ep - (slAtrMult * atrEntry)
    tp = ep + (slAtrMult * rrMult * atrEntry)
    strategy.exit("LX", from_entry="LONG", stop=sl, limit=tp)

plotshape(longSignal, "Long", shape.triangleup, location.belowbar, color.new(color.lime, 0), size=size.small)
